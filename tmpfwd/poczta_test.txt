<?php
// Poczta TD — manual check from SSH or a one-off cron:  /usr/local/bin/php8.x poczta_test.php kadry
// Sends one FICTIONAL message and prints only the HTTP status:
//   200 = the key and the address are right (the portal answers "wylaczona", "do_analizy" or "duplikat")
//   403 = wrong key in ~/.poczta_key      0 = no connection / no TLS      -1 = key file missing or unknown mailbox
require __DIR__ . '/poczta_lib.php';
$box = isset($argv[1]) ? $argv[1] : 'kadry';
$tmp = POCZTA_DIR . '/test.' . getmypid() . '.eml';
file_put_contents($tmp, "From: Test Fikcyjny <test@firma-przyklad.example>\r\nTo: " . $box . "@td-group.pl\r\nSubject: Test przekazania (fikcyjny)\r\n"
    . "Date: " . date('r') . "\r\nMessage-ID: <fwd-test-" . date('Ymd') . "@portal.invalid>\r\nAuto-Submitted: auto-generated\r\n\r\nTo jest fikcyjna wiadomosc testowa.\r\n");
echo 'curl: ' . (function_exists('curl_init') ? 'tak' : 'nie (uzyje strumieni PHP)') . "\n";
echo 'klucz: ' . (poczta_key() !== '' ? 'jest' : 'BRAK ' . POCZTA_KEY_FILE) . "\n";
echo 'HTTP ' . poczta_send($box, $tmp) . "\n";
@unlink($tmp);
