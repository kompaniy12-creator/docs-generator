<?php
// Poczta TD — shared part of the two forwarders (pipe and Maildir cron).
// Sends ONE raw message to the portal webhook. Never prints anything, never throws.
// Lives outside public_html, e.g. /home/host552333/poczta_fwd/poczta_lib.php

define('POCZTA_URL', 'https://dpfxwkxpzqqjtmgqwozw.supabase.co/functions/v1/poczta?action=odbierz&skrzynka=');
define('POCZTA_MAX', 15 * 1024 * 1024);          // bigger messages are left to the portal's own fallback check
define('POCZTA_DIR', __DIR__);                    // spool/, log and state files live next to the scripts
define('POCZTA_KEY_FILE', dirname(__DIR__) . '/.poczta_key');   // /home/host552333/.poczta_key (chmod 600)

function poczta_boxes() { return array('kadry', 'ksiegowosc'); }

function poczta_log($line) {
    $f = POCZTA_DIR . '/poczta.log';
    if (@filesize($f) > 512 * 1024) { @rename($f, $f . '.1'); }   // small: two files of 512 kB at most
    @file_put_contents($f, date('Y-m-d H:i:s') . ' ' . $line . "\n", FILE_APPEND | LOCK_EX);
}

function poczta_key() {
    $k = @file_get_contents(POCZTA_KEY_FILE);
    return $k === false ? '' : trim($k);
}

// Returns the HTTP status (0 = no connection). The log gets numbers only — never a subject or an address.
function poczta_send($box, $file) {
    $key = poczta_key();
    if ($key === '' || !in_array($box, poczta_boxes(), true)) { return -1; }
    $size = @filesize($file);
    if ($size === false || $size < 16) { return -2; }
    if ($size > POCZTA_MAX) { return 413; }
    $url = POCZTA_URL . $box;
    $headers = array('Content-Type: message/rfc822', 'X-Poczta-Key: ' . $key, 'Expect:');
    $body = @file_get_contents($file);              // read-only; at most 15 MB
    if ($body === false) { return -2; }
    if (function_exists('curl_init')) {
        $ch = curl_init($url);
        curl_setopt_array($ch, array(
            CURLOPT_POST => true, CURLOPT_POSTFIELDS => $body, CURLOPT_HTTPHEADER => $headers,
            CURLOPT_RETURNTRANSFER => true, CURLOPT_CONNECTTIMEOUT => 5, CURLOPT_TIMEOUT => 20,
            CURLOPT_SSL_VERIFYPEER => true, CURLOPT_SSL_VERIFYHOST => 2, CURLOPT_FOLLOWLOCATION => false,
        ));
        @curl_exec($ch);
        $code = (int) curl_getinfo($ch, CURLINFO_HTTP_CODE);
        curl_close($ch);
        return $code;
    }
    // no curl extension: plain PHP streams
    $ctx = stream_context_create(array(
        'http' => array('method' => 'POST', 'header' => implode("\r\n", $headers), 'content' => $body, 'timeout' => 20, 'ignore_errors' => true),
        'ssl' => array('verify_peer' => true, 'verify_peer_name' => true),
    ));
    $res = @file_get_contents($url, false, $ctx);
    $code = 0;
    if (isset($http_response_header[0]) && preg_match('~ (\d{3}) ~', $http_response_header[0] . ' ', $m)) { $code = (int) $m[1]; }
    return $code;
}

// 2xx = taken (also "duplicate" and "mailbox switched off"); 400/413/422 = will never be taken: both are final.
// Everything else (no connection, 403, 429, 5xx) is worth another try later.
function poczta_final($code) {
    return ($code >= 200 && $code < 300) || $code === 400 || $code === 413 || $code === 422;
}
